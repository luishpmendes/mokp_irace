# NS-BRKGA Ablation Study — Stage 3: Multiple Populations

# Unlock: num_populations >= 2, exchange_interval > 0, num_exchange_individuals tuned.
# Still no path relinking, no shaking, no reset.
#
# Focused retuning (round 2): each range is restricted to the region held by
# the round-1 surviving ROULETTE configurations on this problem (p10-p90 band
# where they were concentrated; interquartile band, extended to the a-priori
# value, where they were diffuse). Crossover is fixed to ROULETTE and path
# relinking (stages 4-6) to the binary-search IPR.

population_size_factor   "--population-size-factor "    i   (75, 125)
min_elites_percentage    "--min-elites-percentage "     r   (0.05, 0.45)
max_elites_percentage    "--max-elites-percentage "     r   (0.30, 0.60)
mutation_probability     "--mutation-probability "      r   (0.10, 0.90)
mutation_distribution    "--mutation-distribution "     r   (25.00, 55.00)
num_total_parents        "--num-total-parents "         i   (2, 3)
num_elite_parents        "--num-elite-parents "         i   (1, 2)
# CONSTANT=0, CUBIC=1, EXPONENTIAL=2, LINEAR=3, LOGINVERSE=4, QUADRATIC=5, SQRT=6
# Restricted to the steep-bias regime kept in round 1 (CUBIC, QUADRATIC)
# plus the a-priori SQRT
bias_type                "--bias-type "                 c   (1, 5, 6)
# NONE=0, AVERAGE_DISTANCE_TO_CENTROID=1, AVERAGE_DISTANCE_BETWEEN_ALL_PAIRS=2, POWER_MEAN_BASED=3
# Restricted: ALL_PAIRS dropped (O(e^3 n) per generation, not kept in round 1);
# POWER_MEAN_BASED dropped (degenerate on long chromosomes)
diversity_type           "--diversity-type "            c   (0, 1)
# ROULETTE=0, GEOMETRIC=1
# Fixed: ROULETTE only (focused retuning)
crossover_type           "--crossover-type "            c   (0)
num_populations          "--num-populations "           i   (3, 5)
exchange_interval        "--exchange-interval "         i   (100, 650)
num_exchange_individuals "--num-exchange-individuals "  i   (30, 70)
# Fixed: no path relinking, no shaking, no reset
pr_interval              "--pr-interval "               c   (0)
shake_interval           "--shake-interval "            c   (0)
reset_interval           "--reset-interval "            c   (0)

[global]
digits = 2

[forbidden]
(min_elites_percentage >= max_elites_percentage)
(num_elite_parents > num_total_parents)
(num_populations * num_exchange_individuals >= population_size_factor * 4)
(num_elite_parents > (population_size_factor * 4) * min_elites_percentage)
